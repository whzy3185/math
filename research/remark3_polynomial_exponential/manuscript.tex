\begin{abstract}
设 $A=(a_{ij})\in GL_n(\C)$，$p\not\equiv0$ 与 $g$ 为多项式，$\mu_1,\ldots,\mu_n$ 为正整数。本文研究整函数方程
\[
 \prod_{i=1}^n\left(\sum_{j=1}^n a_{ij}u_{z_j}\right)^{\mu_i}=p e^g.
\]
不对解预设有限阶条件。利用混合偏导图和具有共同例外集的对数导数估计，我们首先证明每个方向导数自动具有多项式--指数形式。每个图连通分量内的指数相同，且由 $g$ 的对应变量块唯一确定。随后把全部整函数解归约为有限次不可约因子分配、常数比例检验和一个乘法约束，并给出恢复解的显式径向积分。每个规范解族至多含 $n$ 个复参数；连通情形在忽略加法常数后只有有限多个解。若 $g$ 非常数，则每个解的阶恰为 $\deg g$；若 $g$ 为常数，则每个解为多项式。取全部 $\mu_i=1$ 得到 Lü 的 arXiv:2605.09585v1 中 Remark 3 所述方程的必要充分分类。$p=1$ 的特殊情形恢复原文的一元积分分块结构，而 $p\notin\C$ 时出现真正的多变量指数相位。$p\equiv0$ 的退化情形另行给出。
\end{abstract}
\noindent\textbf{关键词：}整函数；乘积型偏微分方程；对数导数；代数零因子；有限代数分类。

\section{问题来源、研究范围与记号}
Lü 的预印本\cite{Lu2026}研究 $u_{z_1}\cdots u_{z_n}=e^g$ 的整函数解，其中 $g$ 为多项式。其第 6 页 Remark 3 提出两个延伸：在可逆常系数线性方向下处理右端 $e^g$，以及进一步处理右端 $p e^g$，其中 $p$ 为多项式。第一项可通过原文的换元 (1.6) 归约，第二项的额外困难是各方向导数允许具有代数零因子，连通块内的导数比值不再必须为常数。

本文将权重一并纳入，研究
\begin{equation}\label{eq:original}
 \prod_{i=1}^n(D_i u)^{\mu_i}=p(z)e^{g(z)},\qquad
 D_i=\sum_{j=1}^n a_{ij}\partial_{z_j},\quad A\in GL_n(\C),\quad \mu_i\in\N_{>0}.
\end{equation}
除第\ref{sec:degenerate}节外，始终假设 $p\not\equiv0$。变量数允许 $n=1$。记 $\deg$ 为总次数，常数相位的次数在增长估计中记作 $d=0$。整函数阶用最大模定义
\[
 \rho(f)=\limsup_{r\to\infty}\frac{\log\log\max\{e,M(r,f)\}}{\log r},\qquad
 M(r,f)=\max_{\lVert z\rVert\le r}|f(z)|.
\]
这与标准多变量 Nevanlinna 阶一致；其比较见\cite{Stoll1974}。本文使用 $T(r,f)$ 表示以固定中心计算的球面 Nevanlinna 特征。

\paragraph{研究状态。}
本文给出完整的必要性、充分性及参数化证明，但不把新颖性检索等同于优先权认定。尤其是 Xu--Ding 的 2026 年相关论文\cite{XuDing2026}的全部定理尚未逐项核对；与该文的精确重叠需在投稿前完成审查。本文并未声称结果已经同行评审或经证明助理形式化。

\subsection{线性换元}
令
\begin{equation}\label{eq:change}
 z=A^T w,\qquad v(w)=u(A^T w),\qquad
 P(w)=p(A^T w),\quad G(w)=g(A^T w).
\end{equation}
链式法则逐项给出
\[
 \partial_{w_i}v=\sum_{j=1}^n\frac{\partial z_j}{\partial w_i}u_{z_j}
 =\sum_{j=1}^n a_{ij}u_{z_j}=D_i u.
\]
所以原方程等价于
\begin{equation}\label{eq:standard}
 \prod_{i=1}^n F_i^{\mu_i}=P e^G,\qquad F_i=v_{w_i}.
\end{equation}
反变换是 $u(z)=v(A^{-T}z)$，而不是 $v(A^{-1}z)$。由于 $A$ 可逆，整性、是否为多项式、总次数及增长阶均保持不变。以下首先研究\eqref{eq:standard}。

\subsection{混合偏导图}
\begin{definition}
对\eqref{eq:standard}的整函数解 $v$，以 $I=\{1,\ldots,n\}$ 为顶点集，当 $i\ne j$ 且 $v_{w_iw_j}\not\equiv0$ 时连边。记此无向图为 $\Gamma(v)$，连通分量构成的分划为 $\Pi(v)$。单顶点分量也计入分划。对块 $B$ 记
\[
 M_B=\sum_{i\in B}\mu_i.
\]
$\iota_B w_B$ 表示保留 $B$ 中的坐标、把其余坐标置零的向量。
\end{definition}
若 $i\in B$ 且 $j\notin B$，则 $\partial_{w_j}F_i=0$，故 $F_i$ 只依赖 $w_B$。这一简单观察是分块论证的基础。需要特别区分：图分块取决于\emph{解的混合二阶导数}，而不是仅由 $P$ 的因子关系定义。

\section{主结果：规范形与增长刚性}
\begin{theorem}[全部整函数解的规范形]\label{thm:normal}
设 $P\not\equiv0$ 与 $G$ 为 $\C^n$ 上多项式，$\mu_i$ 为正整数。若 $v$ 是\eqref{eq:standard}的整函数解，则存在分划 $\Pi=\Pi(v)$，使得下列结论成立。设
\begin{equation}\label{eq:phase}
 G_0=G(0),\qquad G_B(w_B)=G(\iota_Bw_B)-G_0,\qquad h_B=\frac{G_B}{M_B}.
\end{equation}
首先，$G$ 按该分划可加分离：
\begin{equation}\label{eq:additive}
 G(w)=G_0+\sum_{B\in\Pi}G_B(w_B).
\end{equation}
其次，存在非零多项式 $Q_i\in\C[w_B]$（$i\in B$），满足
\begin{align}
 F_i&=Q_i(w_B)e^{h_B(w_B)},\label{eq:gradient}\
 \prod_{i=1}^n Q_i^{\mu_i}&=e^{G_0}P,\label{eq:productQ}\
 \partial_{w_j}Q_i+Q_i\partial_{w_j}h_B
 &=\partial_{w_i}Q_j+Q_j\partial_{w_i}h_B\quad(i,j\in B).\label{eq:closedQ}
\end{align}
所有解由
\begin{equation}\label{eq:radialmain}
 v(w)=C+\sum_{B\in\Pi}\int_0^1
 e^{h_B(tw_B)}\left(\sum_{i\in B}w_iQ_i(tw_B)\right)\dd t,\qquad C\in\C,
\end{equation}
恢复。反之，对任意分划，若\eqref{eq:phase}--\eqref{eq:closedQ}成立，则\eqref{eq:radialmain}给出\eqref{eq:standard}的整函数解。反向构造时不要求所选分划最细；实际图分量可进一步细分。
\end{theorem}

这里的条件只涉及\emph{多项式恒等式}。第\ref{sec:finite}节进一步说明：多项式 $Q_i$ 也不必作为任意未知函数求解，它们来自 $P$ 的有限次因子分配，剩余未知量仅为常数。

\begin{theorem}[自动增长与精确阶]\label{thm:growth}
在定理\ref{thm:normal}的假设下，若 $G$ 非常数且 $d=\deg G$，则每个整函数解满足 $\rho(v)=d$。若 $G$ 为常数，则 $v$ 为多项式，并且
\begin{equation}\label{eq:degree}
 \deg v\le 1+\left\lfloor\frac{\deg P}{\min_i\mu_i}\right\rfloor.
\end{equation}
特别地，原 Remark 3 的权重全为 $1$ 时，$\deg v\le\deg P+1$。这些结论不以有限阶作为前提。
\end{theorem}

\section{解析预备引理}
本节把增长论证中使用的外部定理及其应用条件明确列出。除标准对数导数估计和基础复解析事实外，所需归约均在下文证明。

\subsection{共同例外集与对数导数}
我们使用 Han--Liu\cite[Theorem 1.1]{HanLiu2025} 的两半径估计。对 $n\ge2$，若亚纯函数 $f$ 非常数且 $f(0)\ne0,\infty$，则 $0<r<R$ 时
\begin{equation}\label{eq:HL}
 m\left(r,\frac{\partial_{w_j}f}{f}\right)
 \le\log^+\!\left(\frac{T(R,f)^2 R^{4n-2}}{(R-r)^{2n}r^{2n-1}}\right)
 +\log^+\log^+\frac1{|f(0)|}+C_f.
\end{equation}
该估计不要求有限阶。经典多变量对数导数背景见 Vitter\cite{Vitter1977}。下文将\eqref{eq:HL}转化成对所有 $F_i$ 同时成立的同半径估计，而不预先假设不同 $F_i$ 的增长可比。

\begin{lemma}[共同增长半径]\label{lem:borel}
设 $S:[1,\infty)\to[1,\infty)$ 连续且单调不减。存在有限 Lebesgue 测度集 $E$，使得对 $r\notin E$，
\[
 S\bigl(r+1/S(r)\bigr)\le2S(r).
\]
\end{lemma}
\begin{proof}
对每个整数 $k\ge0$，令 $r_{k+1}$ 为 $S$ 首次达到 $2^{k+1}$ 的半径；若未达到，则忽略该指标。若 $2^k\le S(r)<2^{k+1}$ 且所需不等式失败，则
\[
 r<r_{k+1}\le r+1/S(r)\le r+2^{-k}.
\]
所以失败集合包含于 $\bigcup_{k\ge0}[r_{k+1}-2^{-k},r_{k+1}]$，其测度至多 $\sum_{k\ge0}2^{-k}=2$。若某些阈值在定义域左端已经达到，相应区间截取到 $[1,\infty)$ 即可。证毕。
\end{proof}

\begin{lemma}[代数零因子的对数导数控制]\label{lem:log}
设 $F_1,\ldots,F_n$ 为非零整函数，所有 $F_i$ 的零因子均由某一非零多项式 $P$ 的零因子控制。选取各 $F_i$ 都不为零的共同中心，并记
\[
 S(r)=1+\sum_{i=1}^n T(r,F_i).
\]
对 $n\ge2$，存在共同的有限测度例外集 $E$，使所有非零对数导数 $L_{ij}=\partial_{w_j}F_i/F_i$ 都满足
\begin{equation}\label{eq:logbound}
 T(r,L_{ij})=O\bigl(\log S(r)+\log r\bigr),\qquad r\notin E.
\end{equation}
\end{lemma}
\begin{proof}
应用引理\ref{lem:borel}，在其补集取 $R=r+1/S(r)$。则 $R\le r+1$ 且 $T(R,F_i)\le S(R)\le2S(r)$。把它代入\eqref{eq:HL}，得到所有非常数 $F_i$ 的共同估计
\[
 m(r,L_{ij})=O(\log S(r)+\log r).
\]
常数 $F_i$ 的导数为零，不产生非零 $L_{ij}$。局部若 $F_i=t^k U$，其中 $U$ 是全纯单位，则
\[
 \frac{\partial_{w_j}F_i}{F_i}=k\frac{\partial_{w_j}t}{t}+\frac{\partial_{w_j}U}{U}.
\]
因此对数导数在约化零除子上至多有一阶极点，不会在 $P=0$ 以外产生极点。故
\[
 N(r,L_{ij})\le N(r,1/P)+O(1)=O(\log r).
\]
加上邻近函数估计即得\eqref{eq:logbound}。可通过先平移到共同非零点来满足\eqref{eq:HL}的中心条件；平移不改变待证的次数及增长阶结论。
\end{proof}

\subsection{多项式零因子的分配}
\begin{lemma}[有限除子分解]\label{lem:divisor}
设非零整函数 $F_i$ 满足 $\prod_i F_i^{\mu_i}=Pe^G$。固定 $P$ 在 $\C[w]$ 中的分解
\begin{equation}\label{eq:factorP}
 P=p_*\prod_{\nu=1}^s\varphi_\nu^{e_\nu},\qquad p_*\in\C^*,
\end{equation}
其中 $\varphi_\nu$ 两两不相伴且不可约。则存在非负整数 $e_{i\nu}$，满足
\begin{equation}\label{eq:allocation}
 \sum_i\mu_i e_{i\nu}=e_\nu,
\end{equation}
以及整函数 $q_i$，使
\[
 F_i=R_i e^{q_i},\qquad R_i=\prod_\nu\varphi_\nu^{e_{i\nu}}.
\]
若进一步 $T(r,F_i)=O(r^d+\log r)$，则 $q_i$ 为次数至多 $d$ 的多项式；$d=0$ 时 $q_i$ 为常数。
\end{lemma}
\begin{proof}
复不可约代数超曲面的光滑开稠密部分在解析拓扑中连通；去除它与其他分量的交也保持连通。这里使用的是复代数簇的代数连通性与解析连通性的一致性，以及光滑复流形去除真解析子集后仍连通的标准事实，参见\cite{GR1965,Achinger2025}。在该光滑部分取横截参数 $t=\varphi_\nu$，由乘积恒等式，每个 $F_i$ 局部为 $t^{e_{i\nu}}$ 乘全纯单位，且指数满足\eqref{eq:allocation}。指数局部常值，故在该连通部分为固定整数。

令 $R_i$ 如上。商 $F_i/R_i$ 及其倒数在去除超曲面奇点集和不同分量交集后均全纯；剩余集合的复余维至少为 $2$。由 Hartogs 延拓，两者全纯延拓到 $\C^n$，且乘积仍为 $1$。$n=1$ 时仅需通常的可去奇点定理。因此 $F_i/R_i$ 是无零整函数。$\C^n$ 单连通，故存在全纯对数 $q_i$。这一步不是在有零点的 $F_i$ 上直接取对数。

现在假设特征增长界。由于 $R_i$ 是多项式，
\[
 T(r,e^{q_i})\le T(r,F_i)+T(r,1/R_i)+O(1)=O(r^d+\log r).
\]
对 $\log^+|e^{q_i}|$ 应用球上的次调和 Poisson 上界，得到
\[
 \max_{\lVert w\rVert\le r}\Re q_i(w)\le C\,T(2r,e^{q_i})+C=O(r^d+\log r).
\]
在每条过原点的复直线上应用 Borel--Carathéodory 不等式，并把半径扩大一倍，可得
\[
 \max_{\lVert w\rVert\le r}|q_i(w)-q_i(0)|=O(r^d+\log r).
\]
多变量 Cauchy 估计说明所有总次数 $>d$ 的 Taylor 系数为零。$d=0$ 时任意正阶系数均为零。因此 $q_i$ 为所述多项式。若增长最初以平移后的中心计算，同一论证先在该中心展开，再平移回来即可。
\end{proof}

\begin{lemma}[指数刚性及变量支撑]\label{lem:rigidity}
以下两条成立。
\begin{enumerate}[label=(\roman*)]
\item 若多项式 $q$ 满足 $e^q\in\C(w_1,\ldots,w_n)$，则 $q$ 为常数。
\item 若非零多项式 $R$ 与多项式 $q$ 满足 $\partial_{w_j}(Re^q)=0$，则 $\partial_{w_j}R=\partial_{w_j}q=0$。
\end{enumerate}
\end{lemma}
\begin{proof}
(i) 取一条一般复仿射直线，使 $q$ 的限制仍保留非零最高次项，且有理函数限制定义良好。若 $q$ 非常数，其指数在该直线的无穷远处有本性奇点，不可能是有理函数。因此 $q$ 必为常数。

(ii) 恒等式等价于 $\partial_{w_j}R+R\partial_{w_j}q=0$。若 $\partial_{w_j}q\ne0$，则 $\deg(R\partial_{w_j}q)\ge\deg R$，而 $\deg(\partial_{w_j}R)\le\deg R-1$（或该导数为零），矛盾。所以 $\partial_{w_j}q=0$，进而 $\partial_{w_j}R=0$。
\end{proof}

\section{自动有限增长的证明}\label{sec:growthproof}
先证明最关键的、不能作为假设使用的增长结论。

\begin{proposition}\label{prop:estimate}
若 $v$ 满足\eqref{eq:standard}，且 $d=\deg G$（常数取 $d=0$），则每个 $F_i$ 满足
\begin{equation}\label{eq:finitebound}
 T(r,F_i)=O(r^d+\log r).
\end{equation}
\end{proposition}
\begin{proof}
当 $n=1$ 时，$F_1^{\mu_1}=Pe^G$，直接使用 $T(r,F_1^{\mu_1})=\mu_1T(r,F_1)+O(1)$ 即得。以下设 $n\ge2$。

因为 $Pe^G\not\equiv0$，每个 $F_i$ 非零。各 $F_i$ 的零因子由 $P$ 控制，所以引理\ref{lem:log}适用。以下所有估计先在其共同例外集的补集成立。

若 $i,j$ 是 $\Gamma(v)$ 的一条边，由混合偏导相等，
\[
 F_i L_{ij}=\partial_{w_j}F_i=\partial_{w_i}F_j=F_j L_{ji}\not\equiv0.
\]
因而作为亚纯恒等式，
\begin{equation}\label{eq:edge}
 \frac{F_i}{F_j}=\frac{L_{ji}}{L_{ij}},\qquad
 T\left(r,\frac{F_i}{F_j}\right)=O(\log S(r)+\log r).
\end{equation}
沿同一连通分量内的一条有限路径连乘，得到对所有 $i,k\in B$ 都成立的同样估计。这里没有把仅对 $F_i$ 小的量擅自视为对 $F_j$ 也小。

令 $H_B=\prod_{i\in B}F_i^{\mu_i}$。因为 $F_i$ 只依赖 $w_B$，$H_B$ 也如此。选取 $P(a)\ne0$ 的点 $a$。固定块外坐标为 $a$，得
\begin{equation}\label{eq:slice}
 H_B(w_B)=\frac{P(w_B,a_{B^c})e^{G(w_B,a_{B^c})}}
 {\prod_{C\in\Pi,\ C\ne B}H_C(a_C)}.
\end{equation}
分母非零常数，切片多项式不是零多项式，因为它在 $a_B$ 处非零。故
\[
 T(r,H_B)=O(r^d+\log r).
\]
对任意 $k\in B$，使用恒等式
\begin{equation}\label{eq:weightedidentity}
 F_k^{M_B}=H_B\prod_{i\in B}\left(\frac{F_k}{F_i}\right)^{\mu_i},
\end{equation}
以及特征函数的乘积和倒数不等式，得到
\[
 M_BT(r,F_k)\le C(r^d+\log r)+C(\log S(r)+\log r).
\]
单顶点分量同样适用，此时比值乘积恒为 $1$。对所有 $k$ 求和并调整常数，得到
\begin{equation}\label{eq:closure}
 S(r)\le C_1(r^d+\log r)+C_2\log S(r)+C_3.
\end{equation}
由 $C_2\log x\le x/2+C_4$，将右端的对数项吸收，得
\[
 S(r)=O(r^d+\log r),\qquad r\notin E.
\]
最后，$E$ 测度有限，所以充分大的每个区间 $[r,r+1]$ 都包含 $E$ 外的点 $s$：否则尾部测度至少为 $1$。由 $S(r)\le S(s)$，增长界延伸到所有充分大的 $r$。这就证明\eqref{eq:finitebound}。
\end{proof}

\begin{remark}
\eqref{eq:closure}是整个推广的闭合步骤。代数零因子只在 $N(r,L_{ij})$ 中贡献 $O(\log r)$，不会破坏该不等式。因此允许多项式因子 $P$ 不需要对解预加有限阶条件。
\end{remark}

\section{规范形的必要性与充分性}
\subsection{必要性}
\begin{proof}[定理\ref{thm:normal}必要性的证明]
由命题\ref{prop:estimate}和引理\ref{lem:divisor}，
\[
 F_i=R_i e^{q_i},\qquad R_i,q_i\in\C[w],\quad R_i\ne0,\quad\deg q_i\le d.
\]
令 $\widetilde q_i=q_i-q_i(0)$，$\widetilde Q_i=e^{q_i(0)}R_i$，于是 $\widetilde q_i(0)=0$，且 $F_i=\widetilde Q_i e^{\widetilde q_i}$。

若 $i,j$ 为图的一条边，则
\begin{align*}
 (\partial_{w_j}\widetilde Q_i+\widetilde Q_i\partial_{w_j}\widetilde q_i)e^{\widetilde q_i}
 =(\partial_{w_i}\widetilde Q_j+\widetilde Q_j\partial_{w_i}\widetilde q_j)e^{\widetilde q_j}\not\equiv0.
\end{align*}
括号中的量都是非零多项式。因此 $e^{\widetilde q_i-\widetilde q_j}$ 是有理函数。由引理\ref{lem:rigidity}(i)，两指数之差为常数；再由原点归一化可知它们相等。沿路径传播，得到每个 $B$ 内的共同相位 $h_B$，满足 $h_B(0)=0$。

对 $i\in B$、$j\notin B$，有 $\partial_{w_j}F_i=0$，引理\ref{lem:rigidity}(ii)说明 $\widetilde Q_i$ 与 $h_B$ 都不依赖 $w_j$。记 $Q_i=\widetilde Q_i\in\C[w_B]$。代入乘积方程得
\[
 \left(\prod_iQ_i^{\mu_i}\right)e^{\sum_B M_Bh_B}=Pe^G.
\]
所以 $e^{\sum_BM_Bh_B-G}$ 为有理函数；指数必须为常数。在原点计算指数常数得到 $-G_0$，从而
\[
 \sum_BM_Bh_B=G-G_0,\qquad \prod_iQ_i^{\mu_i}=e^{G_0}P.
\]
这一步不需要 $P(0)\ne0$，因为计算的是已经证明为常数的多项式指数，而不是在原点除以 $P$。将块外坐标置零，利用每个 $h_C(0)=0$，得到
\[
 M_Bh_B(w_B)=G(\iota_Bw_B)-G_0.
\]
于是\eqref{eq:phase}和\eqref{eq:additive}成立。混合偏导相等给出\eqref{eq:closedQ}。恢复公式将在下面的原函数引理中证明。
\end{proof}

\subsection{全局原函数与充分性}
\begin{lemma}[径向积分原函数]\label{lem:primitive}
设 $f_1,\ldots,f_m$ 是 $\C^m$ 上整函数且 $\partial_jf_i=\partial_if_j$。则
\[
 V(w)=\int_0^1\sum_{i=1}^m w_i f_i(tw)\dd t
\]
是整函数，$V(0)=0$，并满足 $\partial_jV=f_j$。
\end{lemma}
\begin{proof}
被积函数及其任意复偏导在 $[0,1]\times K$ 上一致有界，其中 $K$ 为任意紧集，因此可以在积分号下求复导数。计算
\begin{align*}
 \partial_jV(w)
 &=\int_0^1\left[f_j(tw)+t\sum_iw_i\partial_jf_i(tw)\right]\dd t\\
 &=\int_0^1\left[f_j(tw)+t\sum_iw_i\partial_if_j(tw)\right]\dd t
 =\int_0^1\frac{\dd}{\dd t}\bigl(tf_j(tw)\bigr)\dd t=f_j(w).
\end{align*}
整性与原点值随之成立。
\end{proof}

\begin{proof}[定理\ref{thm:normal}充分性及恢复公式的证明]
对每个块取 $f_i=Q_i e^{h_B}$。\eqref{eq:closedQ}使该块的一形式闭合，引理\ref{lem:primitive}因此给出其全局原函数。块外偏导为零，所以把所有块原函数相加，所得函数的第 $i$ 个偏导恰为 $Q_i e^{h_B}$。利用\eqref{eq:additive}和\eqref{eq:productQ}，
\[
 \prod_i(v_{w_i})^{\mu_i}=\left(\prod_iQ_i^{\mu_i}\right)e^{\sum_BM_Bh_B}
 =e^{G_0}P\,e^{G-G_0}=Pe^G.
\]
反过来，原解与此积分构造具有相同梯度，故在连通的 $\C^n$ 上只相差一个常数。定理得证。
\end{proof}

\subsection{精确增长阶}
\begin{proof}[定理\ref{thm:growth}的证明]
若 $G$ 为常数，\eqref{eq:phase}给出所有 $h_B=0$，因此各 $F_i=Q_i$ 为多项式，径向积分说明 $v$ 为多项式。非零多项式的总次数在乘积下相加，故
\[
 \deg P=\sum_i\mu_i\deg F_i.
\]
对非常数多项式 $v$，$\deg v=1+\max_i\deg F_i$，从而得到\eqref{eq:degree}。$Pe^G\ne0$ 保证 $v$ 不是常数。

若 $d=\deg G\ge1$，由\eqref{eq:phase}，每个 $\deg h_B\le d$。径向积分直接给出
\[
 |v(w)|\le C(1+\lVert w\rVert)^{D+1}\exp(C(1+\lVert w\rVert)^d),\qquad D=\deg P,
\]
所以 $\rho(v)\le d$。又因 $G-G_0=\sum_BM_Bh_B$ 的次数为 $d$，至少一个 $h_B$ 的次数为 $d$。该块任一 $F_i=Q_i e^{h_B}$ 的阶恰为 $d$：选取使相位最高次项非零且 $Q_i$ 限制非零的一般复直线，并在适当方向取半径，便得到匹配的指数增长下界。另一方面，Cauchy 估计给出
\[
 M(r,F_i)\le C r^{-1}M(2r,v),
\]
故 $\rho(F_i)\le\rho(v)$。因此 $\rho(v)=d$。
\end{proof}

\section{有限代数分类：不再求解未知函数}\label{sec:finite}
定理\ref{thm:normal}可以改写为完全有限的数据检验。固定分解\eqref{eq:factorP}，其中 $p_*$ 是所选不可约因子规范化后的非零常数，\emph{不是} $P(0)$。

\begin{definition}[可接受的有限数据]\label{def:data}
一组数据包括分划 $\Pi$ 和非负整数分配 $e_{i\nu}$。要求：
\begin{enumerate}[label=(D\arabic*)]
\item $G$ 满足\eqref{eq:additive}，并按\eqref{eq:phase}定义 $h_B$。
\item 每个 $\nu$ 满足\eqref{eq:allocation}；令 $R_i=\prod_\nu\varphi_\nu^{e_{i\nu}}$。
\item 对 $i\in B$，$R_i$ 只依赖 $w_B$。
\item 存在 $\kappa_i\in\C^*$，使对每个 $i,j\in B$，
\begin{equation}\label{eq:linearK}
 \kappa_i A_{ij}=\kappa_j A_{ji},\qquad
 A_{ij}=\partial_{w_j}R_i+R_i\partial_{w_j}h_B.
\end{equation}
\item 在每个块内，以 $A_{ij}\not\equiv0$ 为边的图连通；单顶点块自动满足。由 (D4)，该边关系是对称的。
\end{enumerate}
在每个块中指定最小指标 $b$，将 $\kappa_b$ 规范为 $1$。
\end{definition}

\begin{theorem}[有限数据与全部解]\label{thm:finite}
所有\eqref{eq:standard}的整函数解恰好由可接受数据的下列族给出：
\begin{align}
 V_B(w_B)&=\int_0^1 e^{h_B(tw_B)}\sum_{i\in B}w_i\kappa_iR_i(tw_B)\dd t,\label{eq:VB}\
 v(w)&=C+\sum_{B\in\Pi}\lambda_BV_B(w_B),\label{eq:family}\
 C&\in\C,\quad\lambda_B\in\C^*,\qquad
 \prod_{B\in\Pi}\lambda_B^{M_B}
 =K:=\frac{p_*e^{G_0}}{\prod_i\kappa_i^{\mu_i}}.\label{eq:torus}
\end{align}
可接受数据只有有限多组；每组中规范化的 $\kappa_i$ 唯一。检查 (D4) 只需比较有限多个多项式系数，或者进行边比例与环路一致性检验，不涉及未知整函数的求解。
\end{theorem}
\begin{proof}
分划数有限。对每个固定 $\nu$，由于 $\mu_i>0$，\eqref{eq:allocation}的非负整数解只有有限多个。因此候选 $R_i$ 有限，且相位已经由 $G$ 和分划确定。

先考察 (D4)。若 $A_{ij}$ 与 $A_{ji}$ 恰有一个为零，则不存在非零常数解。若二者均非零，则它们必须为常数倍；若 $A_{ij}=r_{ij}A_{ji}$，则 $\kappa_j=r_{ij}\kappa_i$。给一个根顶点的 $\kappa=1$，沿连通图的生成树传播比例即可确定全部 $\kappa_i$；其他边成立当且仅当每个闭环上的比例乘积为 $1$。这也证明规范化后的唯一性。二者均为零的顶点对不施加比例条件。

若数据可接受，(D4) 使 $V_B$ 具有梯度 $\kappa_iR_i e^{h_B}$。令 $Q_i=\lambda_B\kappa_iR_i$。由\eqref{eq:allocation}和\eqref{eq:torus}，
\[
 \prod_iQ_i^{\mu_i}
 =\left(\prod_B\lambda_B^{M_B}\right)\left(\prod_i\kappa_i^{\mu_i}\right)\frac{P}{p_*}=e^{G_0}P.
\]
因此定理\ref{thm:normal}给出解。其块内边恰为 $A_{ij}\not\equiv0$，故 (D5) 保证 $\Pi$ 正是该解的实际混合偏导分划。

反之，给定任意解，取其实际分划。由定理\ref{thm:normal}，$\prod Q_i^{\mu_i}=e^{G_0}P$，因此唯一分解给出 $Q_i=c_iR_i$，其中 $c_i\ne0$，且指数满足\eqref{eq:allocation}。变量支撑给出 (D3)，闭合条件给出 (D4)，实际图连通性给出 (D5)。每块中所有常数 $c_i$ 相对于规范化 $\kappa_i$ 只有一个共同比例 $\lambda_B$。常数乘积恰给出\eqref{eq:torus}，原函数只差常数。因此没有遗漏任何整函数解。
\end{proof}

\begin{remark}[“有限”的精确含义]
一般解仍可能含非初等的一元或径向积分。这里的完整分类不是承诺把这些积分全部化为初等函数，而是：输入 $P,G$ 后，只有有限个离散数据选择；通过多项式恒等式检验即可确定是否有解；全部自由度只剩显式常数和一个乘法约束。对任意抽象复系数，这是一种有限代数判据；实际算法要求输入系数和不可约因子可以精确运算与判等。不能把浮点分解当作严格的存在性判定。
\end{remark}

\begin{corollary}[参数维数和连通解计数]\label{cor:count}
设 $t=|\Pi|$。每个可接受数据族由一个加法参数和 $t$ 个非零乘法参数组成，后者满足一个方程\eqref{eq:torus}，故每族的复参数维数为 $t\le n$。更准确地，\eqref{eq:torus}是非空的、维数 $t-1$ 的代数群陪集的有限并；其连通分量数为 $\gcd\{M_B:B\in\Pi\}$。

若只考虑 $\Gamma(v)$ 连通的解并忽略加法常数，记 $M=\sum_i\mu_i$，则解数至多
\begin{equation}\label{eq:count}
 M\prod_{\nu=1}^s\#\left\{(a_1,\ldots,a_n)\in\N^n:\sum_i\mu_i a_i=e_\nu\right\}.
\end{equation}
特别地，所有权重为 $1$ 时，上界为
\[
 n\prod_{\nu=1}^s {e_\nu+n-1\choose n-1}.
\]
\end{corollary}
\begin{proof}
由于 $K\ne0$，可以任意指定 $t-1$ 个非零参数，再对最后一个求 $M_B$ 次根，故参数集合非空。对整数向量 $(M_B)$ 作整可逆基变换，将相应的环面单项式映射化为某一坐标的 $q$ 次幂，其中 $q=\gcd(M_B)$。其非零纤维有 $q$ 个分量，每个分量同构于 $(\C^*)^{t-1}$。加法参数贡献一维。映射\eqref{eq:family}是单射：原点值确定 $C$，每个块内非零偏导确定 $\lambda_B$。

当图连通时分划只有一个块。每种因子分配至多给一个规范方向 $\kappa$，而\eqref{eq:torus}成为 $\lambda^M=K$，恰有 $M$ 个非零根。对全部因子分配计数即得\eqref{eq:count}；无权重情形的组合数来自隔板法。
\end{proof}

\begin{corollary}[连通块的直接多项式检验]\label{cor:connected}
若 $\Gamma(v)$ 连通，令 $M=\sum_i\mu_i$，则存在非零多项式 $q_i$，使
\begin{align}
 v_{w_i}&=q_i e^{G/M},&\prod_iq_i^{\mu_i}&=P,\label{eq:connectedform}\
 M(\partial_{w_j}q_i-\partial_{w_i}q_j)
 &=q_jG_{w_i}-q_iG_{w_j}.\label{eq:connectedtest}
\end{align}
反之，任何满足这两组多项式恒等式的数据均通过径向积分给出解；所得图是否连通由混合偏导另行检查。
\end{corollary}
\begin{proof}
由定理\ref{thm:normal}，唯一相位为 $(G-G_0)/M$。令 $q_i=e^{-G_0/M}Q_i$，立即得到\eqref{eq:connectedform}与\eqref{eq:connectedtest}。充分性仍是引理\ref{lem:primitive}。
\end{proof}

\section{Remark 3 的两个方程与特殊退化}
\subsection{右端没有零点：$P=1$}
\begin{corollary}[一元积分分块结构]\label{cor:zero-free}
对无权重方程 $v_{w_1}\cdots v_{w_n}=e^G$，所有整函数解均可写成
\begin{equation}\label{eq:ridge}
 v(w)=C+\sum_{B\in\Pi}f_B(L_B),\qquad L_B=\sum_{i\in B}c_iw_i,\quad c_i\in\C^*,
\end{equation}
其中 $r_B$ 为一元多项式，$r_B(0)=0$，且
\begin{equation}\label{eq:ridgecond}
 f_B'(s)=e^{r_B(s)},\quad f_B(0)=0,\qquad
 G=G_0+\sum_B|B|r_B(L_B),\qquad \prod_{i=1}^n c_i=e^{G_0}.
\end{equation}
反之，这些数据给出全部解。相同结论对正整数权重成立，只需把 $|B|$ 换成 $M_B$、常数乘积换成 $\prod_i c_i^{\mu_i}$。
\end{corollary}
\begin{proof}
\eqref{eq:productQ}使所有 $Q_i$ 为非零常数 $c_i$。闭合条件变为
\[
 c_i\partial_{w_j}h_B=c_j\partial_{w_i}h_B.
\]
因此 $h_B$ 沿线性函数 $L_B$ 的每个核方向的导数都为零。选取线性向量 $b$ 使 $L_B(b)=1$，则任意 $w_B$ 可以写为 $bL_B(w_B)+\xi$，$L_B(\xi)=0$。沿这些核方向积分，得 $h_B(w_B)=h_B(bL_B(w_B))=:r_B(L_B(w_B))$，其中 $r_B$ 为一元多项式且 $r_B(0)=0$。梯度为 $c_i e^{r_B(L_B)}$，故其原函数为\eqref{eq:ridge}。常数与相位的乘积条件来自定理\ref{thm:normal}。反向直接求导即可。
\end{proof}

推论\ref{cor:zero-free}给出了原文 Remark 3 式 (1.8) 省略的具体步骤：先取\eqref{eq:change}，再使用\eqref{eq:ridge}--\eqref{eq:ridgecond}，最后以 $w=A^{-T}z$ 返回原坐标。这里保留了全部常数乘积，不需要选择不明确的对数分支。

若 $G$ 为常数，所有 $r_B=0$，于是
\[
 v=C+\sum_i c_iw_i,\qquad\prod_i c_i^{\mu_i}=e^{G_0}.
\]
特别是 $P=1$、$G=0$、权重全为 $1$ 时恢复原文的仿射刚性结论。本文使用“仿射”而不是省略加法常数的“齐次线性”。

\subsection{带多项式因子：原式 (1.9)}
令所有权重为 $1$。定理\ref{thm:normal}与定理\ref{thm:finite}，连同\eqref{eq:change}，就是原式 (1.9) 的必要充分分类。与无零点情形不同，$Q_i$ 可以不是常数，闭合条件不再强迫相位通过某个线性形式因子化。

在二维连通情形，上述判据尤其简洁：寻找多项式 $q,r$，使
\begin{equation}\label{eq:2d}
 qr=P,\qquad 2(q_y-r_x)=rG_x-qG_y.
\end{equation}
然后取闭一形式 $e^{G/2}(q\dd x+r\dd y)$ 的径向原函数。二维非连通情形则是两个单变量原函数之和，并要求 $G$ 可加分离及 $P$ 的因子能按变量分配。二者合起来覆盖所有二维整函数解。

\subsection{$p\equiv0$ 的全部解}\label{sec:degenerate}
\begin{proposition}
若 $p\equiv0$，则\eqref{eq:original}的整函数解恰为满足某一个 $D_i u\equiv0$ 的整函数。等价地，在 $w=A^{-T}z$ 坐标中，$v$ 是不依赖至少一个坐标的任意整函数。
\end{proposition}
\begin{proof}
整函数环 $\OO(\C^n)$ 是整环。有限个函数的乘积恒为零，必有一个因子恒为零。因此某个 $v_{w_i}=0$，即 $v$ 不依赖 $w_i$。反之显然。$n=1$ 时这只给出常数解；$n\ge2$ 时允许无限维函数自由度及无限阶解。
\end{proof}

\section{完整算例、锐性与适用边界}
\subsection{一个平方自由因子的完整分类}
\begin{example}\label{ex:xyexp}
方程
\[
 u_xu_y=xy\,e^{2xy}
\]
的全部整函数解为 $u=\pm e^{xy}+C$。
\end{example}
\begin{proof}
$G=2xy$ 不可按单变量可加分离，所以实际图必须连通。由\eqref{eq:2d}，$qr=xy$。忽略非零常数后，$(q,r)$ 只有
\[
 (1,xy),\quad(x,y),\quad(y,x),\quad(xy,1)
\]
四种分配。其对应的 $A_{12}=R_{1,y}+xR_1$ 和 $A_{21}=R_{2,x}+yR_2$ 依次为
\[
 (x,\ y+xy^2),\quad(x^2,\ y^2),\quad(1+xy,\ 1+xy),\quad(x+x^2y,\ y).
\]
只有第三对互为非零常数倍。因此 $q=ay,r=ax$，并由 $qr=xy$ 得 $a^2=1$。积分即得结论。特别地，$P=xy$ 已是平方自由多项式，但 $u_x/u_y=y/x$ 非常数，所以解不是某个线性形式的一元函数。
\end{proof}

\subsection{多项式右端的全部二维解族}
\begin{example}
方程 $u_xu_y=xy$ 的全部整函数解为
\[
 u=\pm xy+C\qquad\text{或}\qquad
 u=\frac12\bigl(ax^2+a^{-1}y^2\bigr)+C,\quad a\in\C^*.
\]
\end{example}
\begin{proof}
由定理\ref{thm:growth}，所有解均为多项式。若图连通，在上述四种因子分配中，仅 $(R_1,R_2)=(y,x)$ 的混合导数均非零且相等，常数乘积使共同系数为 $\pm1$。若图非连通，变量支撑迫使 $R_1=x,R_2=y$，常数仅需满足 $c_1c_2=1$。其余分配的交叉导数恰有一个为零，不可能闭合。分别积分即得全部解。
\end{proof}

\begin{example}[无解的相位]
方程 $u_xu_y=e^{xy}$ 没有整函数解。单顶点分划不满足相位可加分离；连通分划下 $R_1=R_2=1$，但 $A_{12}=x/2$ 与 $A_{21}=y/2$ 不成常数比例。因此所有有限数据均被排除。
\end{example}

\begin{example}[加权扩展]
方程 $u_x^2u_y=xy^2 e^{3xy}$ 的全部整函数解为
\[
 u=\omega e^{xy}+C,\qquad\omega^3=1.
\]
相位仍不可分，故图连通。$x$ 的重数 $1$ 只能分配到第二个导数；$y$ 的重数 $2$ 可全部分给第二个导数，或分给第一个导数一次。第一种分配给出不成常数比例的闭合系数，第二种为 $(R_1,R_2)=(y,x)$，闭合迫使共同常数，而权重乘积给出其三次方为 $1$。
\end{example}

\begin{example}[一个真正的分块族]
方程
\[
 u_xu_yu_z=xy\,e^{2xy+z^2}
\]
具有整函数解族
\[
 u=a e^{xy}+b\int_0^z e^{t^2}\dd t+C,\qquad a,b\in\C^*,\quad a^2b=1.
\]
直接求导即可验证。本例用于说明一个二维非线性相位块与一个一维积分块可以共存；本例的直接计算本身不代替一般分类证明。
\end{example}

\subsection{次数界、参数界与假设的必要性}
无权重且常数相位时，次数界 $\deg u\le\deg P+1$ 是锐的：
\[
 u=\frac{w_1^{D+1}}{D+1}+w_2+\cdots+w_n,
 \qquad\prod_i u_{w_i}=w_1^D.
\]
$D=0$ 时按同一公式得到仿射函数。正整数权重时，选择最小权重对应的坐标令其导数为 $w_i^k$、其他导数为 $1$，在 $D=\mu_i k$ 时同样达到\eqref{eq:degree}。参数上界 $n$ 也可达到：$P$ 为非零常数、$G$ 为常数时，仿射解的 $n$ 个斜率满足一个乘积条件，再加一个自由常数。

$A$ 的可逆性不可删去。例如在二维取两行均为 $(1,0)$，方程变为 $u_x^2=1$，而 $u=x+H(y)$ 对任意整函数 $H$ 都成立，允许无限阶。$g$ 的多项式条件也不可直接删去：
\[
 u=e^{y+e^x},\qquad u_xu_y=e^{x+2y+2e^x},
\]
右端的指数为超越整函数，且该解为无限阶。这些边界例子和 $p=0$ 的退化情形说明自动增长结论的假设具有实质作用。

\section{复核方式与结论边界}
配套程序仅执行有限的精确多项式运算：列出给定因子的全部权重分配，枚举变量分划，验证相位分离、支撑、比例和闭合条件，并对显式例子直接求导。程序接受由调用者提供的不可约因子分解；若因子没有在 $\C$ 上完全分解，则不能把所得枚举宣称为完整。本文的回归算例使用已知线性因子，不存在这一歧义。浮点近似未用于证明恒等式。

解析证明的逻辑顺序为：共同对数导数控制，图内比值估计，块乘积切片，增长闭合，有限除子分解，块内相位一致，闭形式积分，有限参数化。特别地，有限阶是第四步的\emph{结论}，不是分解前的隐藏前提。所有实例验证均与一般解析论证分开记录。

综上，Remark 3 的无零点方程可明确写成一元积分的线性分块，而一般多项式因子方程具有更宽的、多变量相位的有限代数分类。该分类仍能逐一生成全部整函数解，而不是仅给出必要条件。作为研究稿，进一步工作是完成与\cite{XuDing2026,ChenHan2022}的全文新颖性比对和独立数学审读；这些文献审查事项不被伪装成已完成的同行认可。
