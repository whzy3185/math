\begin{abstract}
We consider
\[
u_t=\Delta(\varphi(v)u),\qquad v_t=\Delta v+uF(v)
\]
on a smooth bounded domain with no-flux boundary conditions. Positive conserved cell mass together with a uniform $L^2$ control of the cell density yields a coercive mechanism even without a pointwise lower bound for $u$. For every $q\ge2$ an elementary mass-weighted Poincar\'e estimate converts $\int u|v-v_*|^q$ into control of the signal energy. Consequently, quadratic dissipativity gives exponential decay, whereas $(s-v_*)F(s)\le-\beta|s-v_*|^{2+\theta}$ yields the algebraic signal rate $t^{-1/\theta}$. If $u$ is eventually bounded in one finite $L^p$ space with $p>\max\{n,2\}$, the decay propagates to $L^\infty\times W^{1,\infty}$. In the degenerate case every
\[
0<\mu<\frac1\theta\min\left\{1,\frac{2(p-n)}{pn}\right\}
\]
is admissible for the full system. The result treats general dissipative kinetics, including nonzero equilibria and degenerate damping, and applies to production--consumption and superlinear-consumption models. Spatially homogeneous solutions show that the signal exponent $1/\theta$ is optimal.
\end{abstract}

\section{Introduction and main results}
Let $\Om\subset\R^n$, $n\ge1$, be smooth, bounded and connected. We study nonnegative global classical solutions of
\begin{equation}\label{sys}
 u_t=\Delta(\varphi(v)u),\qquad v_t=\Delta v+uF(v),
\end{equation}
subject to
\begin{equation}\label{bc}
 \partial_\nu(\varphi(v)u)=0,\qquad \partial_\nu v=0.
\end{equation}
Set $m:=\int_\Om u_0>0$ and $\ubar:=m/|\Om|$. Then $m$ is conserved.

A recent motivation is the production--consumption law $F(s)=1-\alpha s$ studied by Qin and Zheng \cite{QinZheng2026}, who establish global boundedness under a structural motility condition; for the same reaction with a classical chemotactic flux, Tao and Winkler proved stabilization \cite{TaoWinkler2025}. For direct consumption $F(s)=-s$, however, signal-dependent-motility systems already possess a substantial large-time theory: Li and Zhao proved boundedness and exponential convergence under their hypotheses \cite{LiZhao2021}; Li and Winkler established relaxation for very weak solutions \cite{LiWinkler2023}; Lauren\c{c}ot obtained spatial homogenization in arbitrary dimension \cite{Laurencot2023}; and refined regularity and smooth stabilization were developed by Li and Winkler \cite{LiWinkler2024}. Related stabilization with a logistic cell source appears in \cite{LiWangPan2021}. Superlinear-consumption kinetics have also appeared in global-solution studies with singular motility \cite{ZhangLi2025}; that singular regime is outside Theorem~\ref{thm:full} when the limiting signal is zero. Quantitative convergence rates for classical chemotactic-flux consumption systems were obtained earlier in \cite{ZhangLi2015}; the present mechanism instead concerns the signal-dependent-motility structure and general weighted kinetics.

Thus neither signal-dependent motility nor stabilization for $F(s)=-s$ is new here. The contribution is a common \emph{rate mechanism}: positive mass closes the signal energy for a general dissipative $F$, including nonzero equilibria and superquadratic degeneracy; the damping order determines an exponential/algebraic dichotomy; and only an eventual finite-$L^p$ bound on $u$ is needed to transfer this decay to strong norms. No sign condition on $\varphi'$ is used in this conditional stabilization stage.

Let $C_P$ satisfy $\norm{f-f_\Om}_2\le C_P\norm{\nabla f}_2$.

\begin{lemma}[Mass-weighted coercivity]\label{lem:coerc}
Let $q\ge2$, $\rho\ge0$, $\int_\Om\rho=m>0$ and $\norm{\rho}_2\le K$. Then
\begin{equation}\label{coerc}
\norm f_2^2\le A\norm{\nabla f}_2^2+B_q\left(\int_\Om\rho|f|^q\right)^{2/q}
\end{equation}
for every $f\in H^1(\Om)$ for which the right-hand side is finite, where
\[
A=C_P^2\left(1+\frac{2|\Om|K^2}{m^2}\right),\qquad B_q=2|\Om|m^{-2/q}.
\]
\end{lemma}
The estimate is elementary; its role is to recover unweighted coercivity from positive mass without pointwise positivity of $\rho$.

\begin{theorem}[Weighted-damping dichotomy]\label{thm:weighted}
Let $z$ be a sufficiently regular solution (for instance, a classical solution) of $z_t=\Delta z+\rho(x,t)F(z)$, $\partial_\nu z=0$, on $[T,\infty)$ and remain in a compact interval $I$. Assume $\rho\ge0$, $\int_\Om\rho(\cdot,t)=m>0$, $\sup_{t\ge T}\norm{\rho(t)}_2\le K$, and for some $z_*\in I$, $\beta>0$ and $q\ge2$,
\begin{equation}\label{dissq}
F(z_*)=0,\qquad (s-z_*)F(s)\le-\beta|s-z_*|^q \quad (s\in I).
\end{equation}
If $q=2$, then $\norm{z(t)-z_*}_2\le Ce^{-\lambda(t-T)}$. If $q>2$, then
\begin{equation}\label{rateweighted}
\norm{z(t)-z_*}_2\le C(1+t-T)^{-1/(q-2)}.
\end{equation}
\end{theorem}

\begin{theorem}[Full stabilization from finite $L^p$ control]\label{thm:full}
Assume that $v(x,t)\in I$ for a compact interval $I$, $\varphi\in C^1(I)$, $\min_I\varphi>0$, $F\in C^1(I)$, and for some finite
\[
p>\max\{n,2\},\qquad \sup_{t\ge T}\norm{u(t)}_p\le U_p.
\]
Let $F(v_*)=0$. If $(s-v_*)F(s)\le-\beta|s-v_*|^2$ on $I$, then, for $t\ge T+2$,
\begin{equation}\label{full-exp}
\norm{u(t)-\ubar}_\infty+\norm{v(t)-v_*}_{W^{1,\infty}}
\le Ce^{-\lambda(t-T)}.
\end{equation}
If instead, for some $\theta>0$,
\begin{equation}\label{deg}
(s-v_*)F(s)\le-\beta|s-v_*|^{2+\theta}\quad(s\in I),
\end{equation}
then $\norm{v(t)-v_*}_2=O((1+t-T)^{-1/\theta})$, and for every
\begin{equation}\label{murange}
0<\mu<\frac1\theta\min\left\{1,\frac{2(p-n)}{pn}\right\},
\end{equation}
\begin{equation}\label{full-alg}
\norm{u(t)-\ubar}_\infty+\norm{v(t)-v_*}_{W^{1,\infty}}
=O((1+t-T)^{-\mu}).
\end{equation}
The constants are independent of $t$ and may depend on the fixed data, $U_p$, and the solution at time $T$.
\end{theorem}

For the final bootstrap we isolate the boundary estimate that will be used below. Write
$\mathcal Q_r(x_0,t):=(\Om\cap B_r(x_0))\times(t-r^2,t)$. For a cylinder $\mathcal C=U\times(t_1,t_2)$ we use the mixed norm
\[
\norm{H}_{L^{p,Q}(\mathcal C)}
:=\left(\int_{t_1}^{t_2}\norm{H(\cdot,\tau)}_{L^p(U)}^Q\,d\tau\right)^{1/Q},
\]
so $L^{p,Q}=L_t^Q L_x^p$.

\begin{lemma}[Boundary $L^2$--$L^\infty$ upgrade]\label{lem:local}
Let $J$ be a time interval and let $y$ be a weak energy solution on $\Om\times J$, with
\[
 y\in L^\infty_{\rm loc}(J;L^2(\Om))\cap L^2_{\rm loc}(J;H^1(\Om)),
\]
of
\[
 y_t-\nabla\!\cdot(a\nabla y)=\nabla\!\cdot H,
 \qquad (a\nabla y+H)\cdot\nu=0,
\]
in the weak sense. Assume $a$ is measurable with $0<a_*\le a\le a^*$ and $H\in L^{p,Q}_{\rm loc}(\Om\times J)$. If $p>\max\{n,2\}$, $Q>2$ and $n/p+2/Q<1$, then there is $r_0\in(0,1]$, depending only on $\Om,a_*,a^*,p,Q$, such that for all admissible backward cylinders compactly contained in the time interval $J$,
\begin{equation}\label{localup}
\norm y_{L^\infty(\mathcal Q_{r_0/2})}
\le C\norm y_{L^2(\mathcal Q_{r_0})}
+C\norm H_{L^{p,Q}(\mathcal Q_{r_0})}.
\end{equation}
This holds for every $n\ge1$.
\end{lemma}

\section{Proofs}
\begin{proof}[Proof of Lemma~\ref{lem:coerc}]
With $f_\Om=|\Om|^{-1}\int_\Om f$, H\"older, Cauchy--Schwarz and Poincar\'e give
\[
m|f_\Om|\le m^{1-1/q}\left(\int_\Om\rho|f|^q\right)^{1/q}+KC_P\norm{\nabla f}_2.
\]
Insert this into $\norm f_2^2=\norm{f-f_\Om}_2^2+|\Om||f_\Om|^2$ and use $(a+b)^2\le2a^2+2b^2$.
\end{proof}

\begin{proof}[Proof of Theorem~\ref{thm:weighted}]
Set $w=z-z_*$, $E=\norm w_2^2$, $G=\norm{\nabla w}_2^2$, and $D=\int_\Om\rho|w|^q$. Testing by $w$ gives
\begin{equation}\label{energy}
\frac12E'+G+\beta D\le0,
\end{equation}
whereas Lemma~\ref{lem:coerc} yields $E\le AG+B_qD^{2/q}$. For $q=2$ this gives a spectral gap. Let $q>2$ and put $X=G+\beta D$. If $X\le1$, then $G\le X$, $D\le X/\beta$ and $X\le X^{2/q}$, hence $E\le CX^{2/q}$. If $X\ge1$, compactness of $I$ gives $E\le C\le CX^{2/q}$. Thus $X\ge cE^{q/2}$ and $E'\le-2cE^{q/2}$. Integration proves \eqref{rateweighted}; if $E$ reaches zero, \eqref{energy} keeps it zero.
\end{proof}

\begin{proof}[Proof of Lemma~\ref{lem:local}]
For $n\ge2$, this is Theorem~1.1 of Choi \cite{Choi2016}. In his notation take $A_{ij}=a\delta_{ij}$, all lower-order coefficients and the scalar source equal to zero, and the forcing vector $F=H$. A smooth bounded domain is a Sobolev extension domain. Set $p_1=p$ and $q_1=Q$. Since all lower-order coefficients and the scalar source vanish, the auxiliary exponents $p_0,q_0,p_2,q_2$ in Choi's hypotheses may be chosen arbitrarily large; hence $p_{\min}=p$ and $q_{\min}=Q$, and his condition is exactly $n/p+2/Q<1$. Since the lower-order norm $D$ there is zero, the admissible radius and constant are independent of the terminal time. Powers of the fixed radius are absorbed into $C$ in \eqref{localup}.

It remains to justify $n=1$, because Choi states his theorem for $d\ge2$. Here $p>2$. Since $1/p+2/Q<1$, choose auxiliary exponents $\widehat p\in(2,p)$ and $\widehat Q\in(2,Q)$ such that
\[
\delta:=1-\frac1{\widehat p}-\frac2{\widehat Q}>0.
\]
On a bounded interval, replace Choi's mixed-norm Lemma~2.3 by the one-dimensional Gagliardo--Nirenberg estimate. If $2<P<\infty$ and $4<R<\infty$ satisfy $1/P+2/R=1/2$, then
\begin{equation}\label{1dembed}
\norm h_{L_t^R L_x^P}
\le C\left(\sup_\tau\norm{h(\tau)}_2+\norm{\partial_xh}_{L^2_{x,t}}\right).
\end{equation}
Indeed, $\norm{h(\tau)}_P\le C\norm{h_x(\tau)}_2^{2/R}\norm{h(\tau)}_2^{1-2/R}+C\norm{h(\tau)}_2$, and integration gives \eqref{1dembed}. To see that the iteration itself closes, let $v_j=(y-k_j)_+$ and $E_j(\tau)=\{v_j(\cdot,\tau)>0\}$. Put $\alpha_*:=((\widehat p-2)\widehat Q)/(\widehat p(\widehat Q-2))$. After Cauchy and H\"older, the forcing term in the Caccioppoli inequality satisfies
\[
\begin{aligned}
\iint |H|\eta_j^2|\partial_xv_j|
&\le\varepsilon\iint\eta_j^2|\partial_xv_j|^2\\
&\quad+C_\varepsilon\norm H_{L^{p,Q}}^2
\left(\int |E_j(\tau)|^{\alpha_*}d\tau\right)^{(\widehat Q-2)/\widehat Q}.
\end{aligned}
\]
The cutoff part is estimated in the same way. Combining this with \eqref{1dembed} gives for the standard De Giorgi quantity $Y_j$ the recurrence
\[
Y_{j+1}\le Cb^jY_j^{1+\frac{2\delta}{1+2\delta}},
\qquad b>1.
\]
The exponent is strictly larger than one because $\delta>0$; hence the usual nonlinear iteration yields \eqref{localup}. The boundary measure bound $|\Om\cap B_r(x_0)|\ge cr$ is automatic on an interval. Thus the one-dimensional case is closed without invoking the $d\ge2$ form of Choi's theorem.
\end{proof}

\begin{proof}[Proof of Theorem~\ref{thm:full}]
Since $p>2$, the assumed $L^p$ bound gives a uniform $L^2$ bound for $u$, so Theorem~\ref{thm:weighted} yields the stated $L^2$ signal rates. Put $w=v-v_*$. Since $F(v_*)=0$ and $F\in C^1(I)$, $|F(v)|\le L_F|w|$. Choose $n<r<p$ and $s$ by $1/r=1/p+1/s$. Interpolation with the bounded signal range (and finite-measure embedding when $s<2$) gives, in the degenerate case,
\begin{equation}\label{wrate}
\norm{w(t)}_s=O((1+t-T)^{-\sigma(r)/\theta}),\quad
\sigma(r):=\min\left\{1,\frac{2(p-r)}{pr}\right\}.
\end{equation}
Hence $\norm{uF(v)}_r$ has the same rate. For $t\ge T+1$,
\[
w(t)=e^{\Delta}w(t-1)+\int_{t-1}^t e^{(t-\tau)\Delta}u(\tau)F(v(\tau))\,d\tau.
\]
The standard Neumann estimate \cite{KeLiWang2022}
\[
\norm{\nabla e^{\tau\Delta}f}_\infty
\le C(1+\tau^{-1/2-n/(2r)})\norm f_r,
\quad0<\tau\le1,
\]
has an integrable singularity because $r>n$. Fixed-time smoothing and \eqref{wrate} give $\norm{\nabla v(t)}_\infty=O((1+t-T)^{-\sigma(r)/\theta})$. Together with $|w_\Om|\le|\Om|^{-1/2}\norm w_2$ and $\norm{w-w_\Om}_\infty\le C\norm{\nabla w}_\infty$, this gives the same $W^{1,\infty}$ rate. Letting $r\downarrow n$ yields every $\mu$ in \eqref{murange}; the quadratic case is identical with exponential interpolation.

Set $q_u=u-\ubar$. Using \eqref{bc}, integration by parts gives
\[
\frac12\frac d{dt}\norm{q_u}_2^2
=-\int_\Om\varphi(v)|\nabla q_u|^2
-\int_\Om u\varphi'(v)\nabla q_u\cdot\nabla v.
\]
Since $p>2$ also gives $\sup_{t\ge T}\norm{u(t)}_2<\infty$, Young and Poincar\'e imply
\begin{equation}\label{cellL2}
\frac d{dt}\norm{q_u}_2^2+c\norm{q_u}_2^2
\le C\norm{\nabla v}_\infty^2.
\end{equation}
Thus $q_u$ inherits the same exponential, respectively admissible algebraic, decay in $L^2$.

Finally write
\[
(q_u)_t-\nabla\cdot(a\nabla q_u)=\nabla\cdot B,
\quad a=\varphi(v),\quad B=u\varphi'(v)\nabla v.
\]
Equation \eqref{bc} is exactly $(a\nabla q_u+B)\cdot\nu=0$, while $a$ is uniformly elliptic and $\norm{B(t)}_p\le C\norm{\nabla v(t)}_\infty$. Choose $Q_*=4p/(p-n)$, so $n/p+2/Q_*=(p+n)/(2p)<1$. Lemma~\ref{lem:local} gives, on a fixed cylinder ending at $t$,
\[
\norm{q_u}_{L^\infty(\mathcal Q_{r_0/2})}
\le C\norm{q_u}_{L^2(\mathcal Q_{r_0})}
+C\norm B_{L^{p,Q_*}(\mathcal Q_{r_0})}.
\]
Both terms on the right have the previously established decay, because $r_0$ is fixed and the estimates are uniform on $(t-r_0^2,t)$. A finite spatial covering proves \eqref{full-exp}--\eqref{full-alg}.
\end{proof}

\section{Applications and sharpness}
The following consequences are illustrations of the two branches of Theorem~\ref{thm:full}, rather than independent boundedness results.

\begin{corollary}[Production--consumption]\label{cor:qz}
Let $F(s)=1-\alpha s$, $\alpha>0$. Then $v_*=1/\alpha$ and $(s-v_*)F(s)=-\alpha(s-v_*)^2$. Hence every solution satisfying Theorem~\ref{thm:full} converges exponentially to $(\ubar,1/\alpha)$. In particular, Qin and Zheng's bounded classical solutions \cite{QinZheng2026} satisfy \eqref{full-exp}: their boundary conditions imply \eqref{bc}, the maximum principle gives
\[
0\le v(x,t)\le\max\{\norm{v_0}_\infty,1/\alpha\},
\]
and uniform boundedness supplies the finite-$L^p$ hypothesis. Their structural assumptions on $\varphi$, including the sign of $\varphi'$, remain part of the upstream boundedness theorem and are not removed by this corollary.
\end{corollary}

\begin{corollary}[Superlinear consumption]\label{cor:super}
Under the remaining hypotheses of Theorem~\ref{thm:full}, suppose $I\subset[0,\infty)$ and $F(s)=-s^\gamma$, $\gamma>1$. Then $v_*=0$ and \eqref{deg} holds with $\theta=\gamma-1$. Therefore
\[
\norm{v(t)}_2=O((1+t-T)^{-1/(\gamma-1)}),
\]
and \eqref{full-alg} holds for every
\[
0<\mu<\frac1{\gamma-1}\min\left\{1,\frac{2(p-n)}{pn}\right\}.
\]
This concerns the same superlinear-consumption kinetics as \cite{ZhangLi2025}, but under the regular-motility assumptions of Theorem~\ref{thm:full}; it does not cover the singular motility regime treated in \cite{ZhangLi2025}.
\end{corollary}

\begin{proposition}[Sharp signal exponent]\label{prop:sharp}
Let $F(s)=-\kappa(s-v_*)|s-v_*|^\theta$, $\kappa,\theta>0$. For homogeneous data $u_0\equiv\ubar>0$ and $v_0-v_*\equiv w_0\ne0$ whose signal trajectory stays in the domain of $F$,
\[
|v(t)-v_*|=\left(|w_0|^{-\theta}+\theta\kappa\ubar t\right)^{-1/\theta}.
\]
Hence the \emph{signal} exponent $1/\theta$ cannot in general be improved; no sharpness claim is made for the full-system exponent in \eqref{murange}.
\end{proposition}

\section{Conclusion}
Positive conserved mass together with finite $L^2$ control converts weighted signal dissipation into a closed energy inequality without pointwise positivity of the cell density. The resulting mechanism extends beyond the pure-consumption setting by allowing general dissipative kinetics, nonzero equilibria and degenerate damping, while the finite-$L^p$ hypothesis separates the asymptotic argument from the problem of obtaining global bounded solutions.
